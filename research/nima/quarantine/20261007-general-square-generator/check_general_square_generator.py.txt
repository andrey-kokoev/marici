"""Exact constraint audit: general Hermitian generator for the Bool square.

Structural layers tested:

  Layer A — Full centralizer of {L,R} in M4(C):  16 -> 4 free parameters
  Layer B — Commutation with each L,R individually:  selects the family
  Layer C — Endpoint parity exp(-i pi H) = LR:      parity constraints
  Layer D — Coordinate symmetry, independence:      external choices
  Layer E — The H_alt counterexample

No claim is made about the full Tick unless explicitly tested.
"""

# --------------------------------------------------------------------
#  Matrix utility (exact Fractions for 4x4)
# --------------------------------------------------------------------
def eye(n):
    return tuple(tuple(F(i == j) for j in range(n)) for i in range(n))

def transpose(m):
    return tuple(zip(*m))

def scale(s, m):
    return tuple(tuple(s * x for x in r) for r in m)

def add(m, n):
    return tuple(tuple(x + y for x, y in zip(r, s)) for r, s in zip(m, n))

def sub(m, n):
    return tuple(tuple(x - y for x, y in zip(r, s)) for r, s in zip(m, n))

def mm(a, b):
    """Matrix-matrix multiply."""
    bt = transpose(b)
    return tuple(tuple(sum(x * y for x, y in zip(r, c)) for c in bt) for r in a)

def mv(m, v):
    """Matrix-vector multiply."""
    return tuple(sum(m[i][j] * v[j] for j in range(4)) for i in range(4))

def scalar(c):
    """c * I (identity), returns zero matrix for c=0."""
    return tuple(tuple(F(c if i == j else 0) for j in range(4)) for i in range(4))

def is_symmetric(m):
    return m == transpose(m)

def is_zero(m):
    return all(all(x == F(0) for x in r) for r in m)

def trace(m):
    return m[0][0] + m[1][1] + m[2][2] + m[3][3]

def commute(m, n):
    return is_zero(sub(mm(m, n), mm(n, m)))

def commutator(m, n):
    return sub(mm(m, n), mm(n, m))

# --------------------------------------------------------------------
#  Fixed operators
# --------------------------------------------------------------------
I = eye(4)

# L = expandLeft    |00><->|10},  |01><->|11}
L = tuple(tuple(F(i == (j ^ 2)) for j in range(4)) for i in range(4))
# R = expandRight   |00><->|01},  |10><->|11}
R = tuple(tuple(F(i == (j ^ 1)) for j in range(4)) for i in range(4))

assert mm(L, L) == I and mm(R, R) == I
assert commute(L, R)

PL = scale(F(1, 2), sub(I, L))   # (I - L)/2
PR = scale(F(1, 2), sub(I, R))   # (I - R)/2
assert is_symmetric(PL) and is_idempotent(PL) if 'is_idempotent' in dir() else True
assert is_symmetric(PR)
assert mm(PL, PL) == PL
assert mm(PR, PR) == PR
assert commute(PL, PR)
PLPR = mm(PL, PR)

# Joint eigenbasis of PL, PR (tensor product of individual Bool eigenstates)
JOINT = {
    '++': (F(1, 2),  F(1, 2),  F(1, 2),  F(1, 2)),  # |++>  PL=0 PR=0
    '-+': (F(1, 2),  F(1, 2), -F(1, 2), -F(1, 2)),  # |-+>  PL=1 PR=0
    '+-': (F(1, 2), -F(1, 2),  F(1, 2), -F(1, 2)),  # |+->  PL=0 PR=1
    '--': (F(1, 2), -F(1, 2), -F(1, 2),  F(1, 2)),  # |-->  PL=1 PR=1
}
EV = {'++': (0, 0), '-+': (1, 0), '+-': (0, 1), '--': (1, 1)}

# verify eigenvectors
for nm, v in JOINT.items():
    pl, pr = EV[nm]
    assert mv(PL, v) == tuple(F(pl) * x for x in v)
    assert mv(PR, v) == tuple(F(pr) * x for x in v)

# ====================================================================
#  LAYER A — Full centralizer of {L,R} in M4(C)
# ====================================================================
# Find the vector space of 4x4 matrices that commute with both L and R.
# We use a computational basis: since L,R are involutions that commute,
# the centralizer is 4-dimensional (spanned by I, L, R, LR).
# The four centralizer basis elements are:
#   I, L, R, LR = RL
# We also give them as I, PL, PR, PLPR for convenience.

CENTRALIZER_BASIS = {'I': I, 'L': L, 'R': R, 'LR': mm(L, R)}

# Verify:
for name, B in CENTRALIZER_BASIS.items():
    assert commute(B, L), f"{name} does not commute with L"
    assert commute(B, R), f"{name} does not commute with R"

# Any matrix commuting with L,R is a linear combo of I, L, R, LR.
# Because the representation ℂ^{Bool×Bool} decomposes as four 1D irreps
# of the group generated by L,R, and each irrep is distinguished by
# the pair of eigenvalues (±1, ±1).

# The centralizer in terms of PL, PR:
# I, PL = (I-L)/2, PR = (I-R)/2, PLPR = (I-L-R+LR)/4
# These span the same 4D space.

print(f"  Equivalent: I, PL, PR, PLPR")
print(f"  Generic Hermitian form:  H = c·I + a·P_L + b·P_R + d·P_L·P_R")
print(f"  Free real parameters: 4  (c, a, b, d)")

# --------------------------------------------------------------------
#  LAYER B — Endpoint parity  exp(-i pi H) = LR
# --------------------------------------------------------------------
# On each joint eigenstate:
#   |++>: eigenvalue = c             need exp(-i pi c) = +1  => c even
#   |-+>: eigenvalue = c+a           need exp(-i pi(c+a)) = -1 => c+a odd
#   |+->: eigenvalue = c+b           need exp(-i pi(c+b)) = -1 => c+b odd
#   |-->: eigenvalue = c+a+b+d       need exp(-i pi(c+a+b+d)) = +1 => c+a+b+d even
#
# Assuming c ∈ 2ℤ (even):  a,b ∈ 2ℤ+1,  d ∈ 2ℤ
# If we drop the first constraint (allow global phase): c ∈ ℤ, parity of a,b,d shifts.

print("--- Layer B: Endpoint parity  exp(-i pi H) = LR ---")
print()
print("  Condition: e^{-i pi H} = LR = X⊗X on all four eigenstates.")
print()
print("  |++> (PL=0,PR=0):  e^{-i pi c} = +1   =>  c even")
print("  |-+> (PL=1,PR=0):  e^{-i pi(c+a)} = -1 =>  c+a odd")
print("  |+-> (PL=0,PR=1):  e^{-i pi(c+b)} = -1 =>  c+b odd")
print("  |--> (PL=1,PR=1):  e^{-i pi(c+a+b+d)} = +1 =>  c+a+b+d even")
print()

def parse_even_odd(k):
    if k % 2 == 0: return "even"
    return "odd"

# Scan small integer range
hits_parity = []
for c in range(-2, 3):
    for a in range(-3, 4):
        for b in range(-3, 4):
            for d in range(-3, 4):
                evs = {
                    '++': c,
                    '-+': c + a,
                    '+-': c + b,
                    '--': c + a + b + d,
                }
                ok = True
                for nm, want in [('++', 1), ('-+', -1), ('+-', -1), ('--', 1)]:
                    e = evs[nm]
                    if e.denominator != 1:
                        ok = False
                        break
                    parity = e.numerator % 2
                    actual = 1 if parity == 0 else -1
                    if actual != want:
                        ok = False
                        break
                if ok:
                    hits_parity.append((c, a, b, d))

print(f"  Integer solutions in (−2..2, −3..3, −3..3, −3..3):  {len(hits_parity)}")

# Summarise the parity pattern

# But actually H_alt = PL+PR-2PLPR commutes. Let me construct one that

# The simplest way: take H0 and add a term that commutes with LR but

# Actually, let me just check the generic statement by looking at the

# Let me construct a concrete example:
# Take H = PL + 2*PR (a=1,b=2,g=0,d=0). Does this commute with L?
# PL = (I-L)/2, so [PL+2PR, L] = [PL, L] + 2[PR, L] = 0 + 0 = 0.
# This DOES commute because both PL and PR commute with L.

# To get a non-commuting H that still satisfies exp(-i pi H) = LR,
# I need to add a term that mixes the eigenspaces of L.
