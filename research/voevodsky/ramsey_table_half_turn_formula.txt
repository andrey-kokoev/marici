=LET(
  width,P1,
  span,width-2,
  ids,SEQUENCE(span^3,,0),
  fst,QUOTIENT(ids,span^2)+1,
  mid,MOD(QUOTIENT(ids,span),span)+1,
  lst,MOD(ids,span)+1,
  pos,FILTER(HSTACK(fst,mid+1,lst+2),(fst<mid)*(mid<lst)),
  mirror,LAMBDA(lo,hi,
    AND(INDEX(pos,lo,1)+INDEX(pos,hi,3)=width+1,
        INDEX(pos,lo,2)+INDEX(pos,hi,2)=width+1,
        INDEX(pos,lo,3)+INDEX(pos,hi,1)=width+1)),
  build,LAMBDA(self,lo,hi,
    IF(lo=hi,
       IF(mirror(lo,hi),lo,VSTACK(lo,-lo)),
       IF(mirror(lo,hi),
          IF(hi=lo+1,VSTACK(lo,hi),
             VSTACK(lo,self(self,lo+1,hi-1),hi)),
          VSTACK(lo,self(self,lo+1,hi),-lo)))),
  rowids,build(build,1,ROWS(pos)),
  MAKEARRAY(ROWS(rowids),width,LAMBDA(ii,jj,
    LET(src,INDEX(rowids,ii),
        colnum,IF(src>0,jj,width+1-jj),
        IF(OR(colnum=INDEX(pos,ABS(src),1),
              colnum=INDEX(pos,ABS(src),2),
              colnum=INDEX(pos,ABS(src),3)),"R","B"))))
)
